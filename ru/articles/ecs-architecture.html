    Архитектура ECS

    AWEngine использует паттерн ECS для разделения данных и логики. Все примеры только примеры, и написаны МАКСИАМЛЬНО просто.

    EcsEntity

    • EcsEntity - уникальный контейнер для компонентов. Имеет такие методы как:

      • EcsEntity.AddComponent(EcsComponent component) Добавляет в сущность компонент. Поддерживает цепочку вызовов(EcsEntity.AddComponent(EcsComponent component).AddComponent(EcsComponent component)).

      • EcsEntity.RemoveComponent<T>() where T : EcsComponent Удаляет компонент из сущности. Также поддерживает цепочку вызововю

      • EcsEntity.SwapComponent<T>(EcsComponent S) where T : EcsComponent Меняет компонент T на компонент S. Сделал чтобы 2 раза не писать удалить такой-то компонент и добавить такой-то компонент

      • EcsEntity.GetComponent<T>() where T : EcsComponent Получить компонент T из энтити, чтобы работать с его содержимым.

      • EcsEntity.HasComponent<T>() where T : EcsComponent Возвращает true, если компонент T есть в сущности

    Пример с EcsEntity :

    var hero = new EcsEntity();
hero.AddComponent(new Info { Name = "Главный герой", Description = "Крутой чувак"} );
hero.AddComponent(new Respect());
hero.AddComponent(new Inventory { Space = 10 } );

if (2+2 != 5) 
{
    hero.SwapComponent<Respect>(new Disrespect());
}

if (hero.HasComponent<Respect>())
{
    hero.GetComponent<Inventory>().Space = 20
}
else
{
    hero.RemoveComponent<Inventory>();
}

    EcsComponent

    • EcsComponent - чистые данные (наследуются от EcsComponent).

    Пример с EcsComponent:

    public class Respect : EcsComponent //Можно делать и пустые, если хотите его использовать как маркер, да и вообще делайте че хотите.
{

}

public class Inventory : EcsComponent
{
    public int Space { get; set; }
}

    EcsSystem

    • EcsSystem - игровая логика, обрабатывающая сущности. Регистрируютсля через SystemManager по фазам выполнения (ExecutionPhase).

      • Классу наследнику ОБЯЗАТЕЛЬНО требуется обработать public override void Update(IReadOnlyList<EcsEntity> entities)

    Пример с EcsSystem :

    public class AOEDamage : EcsSystem
{
    public override void Update(IReadOnlyList<EcsEntity> entities)
    {
        var mainHero = entities.FirstOrDefault(s => s.HasComponent<MainHero>());
        foreach(EcsEntity entity in entities)
        {
            if (entity.HasComponent<Enemy>())
            {
                entity.GetComponent<Stats>().HP -= mainHero.GetComponent<Stats>().AOEDamage
            }
        }
    }
}

    WorldContext

    • WorldContext - хранилище всех активных сущностей(EcsEntity) мира. Имеет следующие методы:

      • WorldContext.AddEntity(EcsEntity entity) Добавляет новую сущность в хранилище.

      • WorldContext.Cleanup<T>() where T : EcsComponent Удаляет все сущности, имеющие T компонент(например вы создали компонент Remove, и все сущности имеющие этот компонент будут удалены при вызове этого метода.)

    Пример с WorldContext:

    var wc = new WorldContext();
var hero = new EcsEntity();
var enemy = new EcsEntity();

hero.AddComponent(new MainHero()).AddComponent(new Stats{ HP = 100, Damage = 50});
enemy.AddComponent(new Enemy()).AddComponent(new Stats{ HP = 30, Damage = 15});

wc.AddEntity(hero);
wc.AddEntity(enemy);

var mainHero = wc.Entities.FirstOrDefault(s => s.HasComponent<MainHero>());
var enemy = wc.Entities.FirstOrDefault(s => s.HasComponent<Enemy>());

if(enemy.GetComponent<Stats>().HP - mainHero.GetComponent<Stats>().Damage <= 0)
{
    enemy.AddComponent(new Remove());
}
else
{
    Console.WriteLine("Фу бот не убил");
}

wc.Cleanup<Remove>();

    SystemManager

    • SystemManager - хранилище всех зарегистрированных систем(EcsSystem) по фазам выполнения(ExecutionPhase). Имеет методы:

      • SystemManager.Add(ExecutionPhase phase, EcsSystem system) Добавляет в список систему по определенному типу фазы выполнения. Поддерживает перегрузку и вместо еденичной системы может принять List<EcsSystem>

      • SystemManager.Remove(ExecutionPhase phase, EcsSystem system) Убирает из списка систему по определенному типу фазы выполнения.

      • SystemManager.Update(WorldContext wc, ExecutionPhase phase) Выполняет все зарегестрированные системы по типу фазы выполнения

    Пример с SystemManager:

    public static class GameSystems
{
    public static ExecutionPhase render = new ExecutionPhase();
    public static ExecutionPhase input = new ExecutionPhase();
}

var wc = new WorldContext();
var sm = new SystemManager();
systems.Add(GameSystems.input, new GetButton());
systems.Add(GameSystems.render, new MainMenu());

var ui = new Entity();
ui.AddComponent(new MainMenuUI())
wc.AddEntity(ui);

sm.Update(wc, GameSystems.input);
sm.Update(wc, GameSystems.render);
